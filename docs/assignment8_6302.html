# Create Texas Government Employee Data (TRS - Teacher Retirement System)
create_trs_data <- function(n = 500) {
  
  first_names <- c("John", "Maria", "Robert", "Jennifer", "Michael", "Sarah",
                   "David", "Patricia", "James", "Linda", "Daniel", "Barbara")
  last_names <- c("Smith", "Johnson", "Williams", "Brown", "Jones", "Garcia",
                  "Miller", "Davis", "Rodriguez", "Martinez", "Hernandez", "Lopez")
  
  trs <- tibble(
    record_id = paste0("TRS_", 1:n),
    first_name = sample(first_names, n, replace = TRUE),
    last_name = sample(last_names, n, replace = TRUE),
    dob = sample(seq(as.Date("1950-01-01"), as.Date("1985-12-31"), by = "day"), n),
    years_service = sample(5:40, n, replace = TRUE),
    salary_group = sample(c("Elementary", "Secondary", "Administration"), n, replace = TRUE),
    retirement_date = NA_Date_
  ) %>%
    mutate(
      # Some retirements for older employees
      retirement_date = case_when(
        years_service > 30 ~ as.Date(paste0(2020 + sample(0:4, n, replace = TRUE), 
                                              "-", 
                                              sample(1:12, n, replace = TRUE, prob = rep(1/12, 12)),
                                              "-01")),
        TRUE ~ NA_Date_
      )
    )
  
  return(trs)
}

# Create Education Agency Data (TEA - Texas Education Agency)
create_tea_data <- function(trs, n = 500, num_matches = 100, corruption_rate = 0.2) {
  
  first_names <- c("John", "Maria", "Robert", "Jennifer", "Michael", "Sarah",
                   "David", "Patricia", "James", "Linda", "Daniel", "Barbara",
                   "Jon", "Mary", "Rob", "Jen", "Mike", "Sara")  # Include variations
  last_names <- c("Smith", "Smithe", "Smyth", "Johnson", "Jonson", "Williams", 
                  "Brown", "Jones", "Garcia", "Miller", "Davis", "Rodriguez", 
                  "Martinez", "Hernandez", "Lopez", "Sanchez")
  
  tea <- tibble(
    record_id = paste0("TEA_", 1:n),
    first_name = sample(first_names, n, replace = TRUE),
    last_name = sample(last_names, n, replace = TRUE),
    dob = sample(seq(as.Date("1950-01-01"), as.Date("1985-12-31"), by = "day"), n),
    certification = sample(c("Science", "Math", "English", "Social Studies", "Special Ed"), n, replace = TRUE),
    school_district = sample(paste0("District_", 1:50), n, replace = TRUE),
    hire_date = sample(seq(as.Date("1990-01-01"), as.Date("2020-12-31"), by = "day"), n)
  )
  
  # Make the first num_matches TEA records true matches of TRS_1..TRS_num_matches.
  # The remaining TEA records are unrelated people (true non-matches).
  tea$first_name[1:num_matches] <- trs$first_name[1:num_matches]
  tea$last_name[1:num_matches] <- trs$last_name[1:num_matches]
  tea$dob[1:num_matches] <- trs$dob[1:num_matches]

  # Introduce data quality issues in the matched records only
  # About 20% of matched records have corruption or variation
  corruption_indices <- sample(1:num_matches, size = round(num_matches * corruption_rate), replace = FALSE)
  
  for (idx in corruption_indices) {
    flip <- sample(1:3, 1)
    
    if (flip == 1) {
      # Introduce name variation
      tea$first_name[idx] <- abbreviate(tea$first_name[idx], minlength = 3)
    } else if (flip == 2) {
      # Introduce small typo in last name
      last_chars <- nchar(tea$last_name[idx])
      if (last_chars > 1) {
        typo_pos <- sample(1:last_chars, 1)
        wrong_char <- sample(letters[!(letters %in% unlist(strsplit(tea$last_name[idx], "")))], 1)
        tea$last_name[idx] <- paste0(
          substr(tea$last_name[idx], 1, typo_pos - 1),
          wrong_char,
          substr(tea$last_name[idx], typo_pos + 1, last_chars)
        )
      }
    } else {
      # Shift DOB slightly (data entry error)
      tea$dob[idx] <- tea$dob[idx] + sample(c(-1, 1), 1)
    }
  }
  
  return(tea)
}

# Create a subset of KNOWN matches for validation
create_validation_set <- function(trs, tea, num_matches = 100) {
  
  # Take first 100 records from TRS
  trs_subset <- trs %>% slice(1:num_matches)
  
  # These are the "true" matches in TEA
  true_matches <- tea %>%
    slice(1:num_matches) %>%
    rename_with(~paste0("tea_", .), everything()) %>%
    rename(tea_record_id = tea_record_id)
  
  # Create ground truth
  validation <- tibble(
    trs_record_id = trs_subset$record_id,
    tea_record_id = true_matches$tea_record_id,
    is_match = TRUE
  )
  
  return(validation)
}

  name %>%
    tolower() %>%              # Convert to lowercase
    str_trim() %>%             # Remove leading/trailing whitespace
    str_squish() %>%           # Remove extra internal whitespace
    str_replace_all("[^a-z ]", "")  # Remove punctuation/special characters
}

deterministic_linkage <- function(data_a, data_b, 
                                  id_col_a = "record_id", 
                                  id_col_b = "record_id") {
  
  log_info("Starting deterministic linkage...")
  
  matches <- tibble()
  remaining_a <- data_a
  remaining_b <- data_b
  
  # RULE 1: Exact match on last name + first name + DOB
  # (Highest confidence)
  log_info("Applying Rule 1: Last name + First name + DOB")
  
  rule1 <- remaining_a %>%
    inner_join(
      remaining_b,
      by = c("last_name_std" = "last_name_std",
             "first_name_std" = "first_name_std",
             "dob_std" = "dob_std"),
      suffix = c("_a", "_b")
    ) %>%
    mutate(
      rule_applied = "Rule_1_FullName_DOB",
      confidence = 0.99
    ) %>%
    select(!!sym(paste0(id_col_a, "_a")), 
           !!sym(paste0(id_col_b, "_b")), 
           rule_applied, confidence)
  
  matches <- bind_rows(matches, rule1)
  
  # Remove matched records
  remaining_a <- anti_join(remaining_a, rule1, by = setNames("record_id_a", id_col_a))
  remaining_b <- anti_join(remaining_b, rule1, by = setNames("record_id_b", id_col_b))
  
  log_info(paste("Rule 1 found", nrow(rule1), "matches"))
  
  # RULE 2: Last name + First initial + DOB
  # (For records where first name varies slightly)
  log_info("Applying Rule 2: Last name + First initial + DOB")
  
  rule2 <- remaining_a %>%
    inner_join(
      remaining_b,
      by = c("last_name_std" = "last_name_std",
             "first_initial" = "first_initial",
             "dob_std" = "dob_std"),
      suffix = c("_a", "_b")
    ) %>%
    mutate(
      rule_applied = "Rule_2_LastName_Initial_DOB",
      confidence = 0.95
    ) %>%
    select(record_id_a, record_id_b, rule_applied, confidence)

  matches <- bind_rows(matches, rule2)
  remaining_a <- anti_join(remaining_a, rule2, by = setNames("record_id_a", id_col_a))
  remaining_b <- anti_join(remaining_b, rule2, by = setNames("record_id_b", id_col_b))
  
  log_info(paste("Rule 2 found", nrow(rule2), "matches"))
  
  # RULE 3: Last name + First name + DOB within 1 day
  # (For data entry errors on DOB)
  log_info("Applying Rule 3: Last name + First name + DOB (±1 day)")
  
  rule3 <- remaining_a %>%
    cross_join(remaining_b, suffix = c("_a", "_b")) %>%
    filter(
      last_name_std_a == last_name_std_b,
      first_name_std_a == first_name_std_b,
      abs(as.numeric(difftime(dob_std_a, dob_std_b, units = "days"))) <= 1
    ) %>%
    select(record_id_a, record_id_b) %>%
    mutate(
      rule_applied = "Rule_3_DOBFlex",
      confidence = 0.90
    )

  if (nrow(rule3) > 0) {
    # Remove duplicates (keep first match only)
    rule3 <- rule3 %>% distinct(record_id_a, .keep_all = TRUE)
    matches <- bind_rows(matches, rule3)
    remaining_a <- anti_join(remaining_a, rule3, by = setNames("record_id_a", id_col_a))
    remaining_b <- anti_join(remaining_b, rule3, by = setNames("record_id_b", id_col_b))
  }
  
  log_info(paste("Rule 3 found", nrow(rule3), "matches"))
  
  result <- list(
    matches = matches,
    remaining_a = remaining_a,
    remaining_b = remaining_b,
    summary = tibble(
      Rule = c("Rule 1", "Rule 2", "Rule 3"),
      Matches = c(nrow(rule1), nrow(rule2), nrow(rule3))
    )
  )
  
  return(result)
}

# Calculate string similarity using Jaro-Winkler
calculate_string_similarity <- function(str1, str2, method = "jw") {
  
  if (is.na(str1) | is.na(str2)) {
    return(NA_real_)
  }
  
  # Jaro-Winkler returns distance (0-1), we want similarity
  1 - stringdist(str1, str2, method = method)
}

# More efficient: Use vectorized stringdistmatrix
calculate_similarity_efficient <- function(data_a_field, data_b_field) {
  
  dist_matrix <- stringdistmatrix(
    data_a_field,
    data_b_field,
    method = "jw",
    useNames = "strings"
  )
  
  # Convert distance to similarity (1 - distance)
  similarity <- 1 - dist_matrix
  
  return(similarity)
}

# ASSESSMENT: PRECISION, RECALL, F1-SCORE
# ============================================================================

assess_linkage_quality <- function(all_matches, validation_set) {
  
  log_info("Assessing linkage quality against validation set...")
  
  our_matches <- all_matches %>%
    mutate(our_match = 1) %>%
    select(trs_record_id, tea_record_id, our_match)
  
  true_matches <- validation_set %>%
    rename(trs_record_id = trs_record_id, 
           tea_record_id = tea_record_id) %>%
    mutate(true_match = 1) %>%
    select(trs_record_id, tea_record_id, true_match)
  
  # Full outer join to get all possibilities
  assessment <- true_matches %>%
    full_join(our_matches, 
              by = c("trs_record_id", "tea_record_id")) %>%
    mutate(
      true_match = replace_na(true_match, 0),
      our_match = replace_na(our_match, 0)
    )
  
  tp <- sum(assessment$true_match == 1 & assessment$our_match == 1)
  fp <- sum(assessment$true_match == 0 & assessment$our_match == 1)
  fn <- sum(assessment$true_match == 1 & assessment$our_match == 0)
  tn <- sum(assessment$true_match == 0 & assessment$our_match == 0)
  
  precision <- if (tp + fp > 0) tp / (tp + fp) else NA_real_
  recall <- if (tp + fn > 0) tp / (tp + fn) else NA_real_
  f1 <- if (!is.na(precision) & !is.na(recall) & precision + recall > 0) {
    2 * (precision * recall) / (precision + recall)
  } else {
    NA_real_
  }
  
  accuracy <- (tp + tn) / nrow(assessment)
  
  results <- list(
    metrics = tibble(
      Metric = c("Precision", "Recall", "F1-Score", "Accuracy"),
      Value = c(precision, recall, f1, accuracy),
      Interpretation = c(
        "Of matches we made, what % were correct?",
        "Of all true matches, what % did we find?",
        "Harmonic mean of precision and recall",
        "What % of all decisions were correct?"
      )
    ),
    confusion_matrix = tibble(
      Metric = c("True Positives", "False Positives", "False Negatives", "True Negatives"),
      Count = c(tp, fp, fn, tn)
    ),
    detailed = assessment
  )
  
  return(results)
}

sensitivity_analysis <- function(all_comparisons, validation_df = validation_set, include_deterministic = FALSE, det_matches = NULL) {
  
  log_info("Running sensitivity analysis on thresholds...")
  thresholds <- seq(0.70, 0.99, by = 0.05)
  results <- tibble()
  
  for (threshold in thresholds) {
    test_matches <- all_comparisons %>%
      filter(match_score >= threshold) %>%
      mutate(our_match = 1) %>%
      select(record_id_a, record_id_b, our_match)
    
    if (include_deterministic && !is.null(det_matches)) {
      test_matches <- bind_rows(
        det_matches %>% select(record_id_a, record_id_b) %>% mutate(our_match = 1),
        test_matches
      ) %>% distinct(record_id_a, record_id_b, .keep_all = TRUE)
    }
    
    # Full validation
    assessment <- validation_df %>%
      rename(record_id_a = trs_record_id, record_id_b = tea_record_id) %>%
      full_join(test_matches, by = c("record_id_a", "record_id_b")) %>%
      mutate(
        is_match = replace_na(is_match, 0),
        our_match = replace_na(our_match, 0)
      )
    
    tp <- sum(assessment$is_match == 1 & assessment$our_match == 1)
    fp <- sum(assessment$is_match == 0 & assessment$our_match == 1)
    fn <- sum(assessment$is_match == 1 & assessment$our_match == 0)
    
    precision <- if (tp + fp > 0) tp / (tp + fp) else 0
    recall <- if (tp + fn > 0) tp / (tp + fn) else 0
    f1 <- if (precision + recall > 0) 2 * (precision * recall) / (precision + recall) else 0
    
    results <- bind_rows(results, tibble(
      threshold = threshold,
      num_matches = nrow(test_matches),
      precision = precision,
      recall = recall,
      f1 = f1
    ))
  }
  return(results)
}

Question: What does the Fellegi-Sunter model estimate that the workshop’s weighted score does not?

In comparison the the default linear scoring th Fellegi-Sunter model uses an Expectation-Maximization (EM) algorithm to the the observed agreement patterns without a training key. The Fellegi-Sunter model is able to get the posterior match probability for each candidate pairs using the priors and the field conditional probabilities of agreement.

Question: Why does the paper treat records with a unique ID differently from records without one?

When a unique identity field is available, a merge is quite easy as the linkages are clean and standardized. Without this identifier field, there is always a degree of uncertainty around if a given record is actually related to the records in the data set to which the merging is happening.

evaluate_threshold_grid <- function(comparisons_df, det_matches_df, validation_df, scheme_label) {
  grid <- seq(0.70, 0.99, by = 0.01)
  map_dfr(grid, function(thresh) {
    prob_m <- comparisons_df %>%
      filter(match_score >= thresh) %>%
      select(trs_record_id = record_id_a, tea_record_id = record_id_b)
    
    comb_m <- bind_rows(
      det_matches_df %>% select(trs_record_id = record_id_a, tea_record_id = record_id_b),
      prob_m
    ) %>% distinct()
    
    assess <- validation_df %>%
      mutate(true_match = 1) %>%
      full_join(comb_m %>% mutate(our_match = 1), by = c("trs_record_id", "tea_record_id")) %>%
      mutate(true_match = replace_na(true_match, 0), our_match = replace_na(our_match, 0))
    
    tp <- sum(assess$true_match == 1 & assess$our_match == 1)
    fp <- sum(assess$true_match == 0 & assess$our_match == 1)
    fn <- sum(assess$true_match == 1 & assess$our_match == 0)
    
    prec <- if (tp + fp > 0) tp / (tp + fp) else 0
    rec <- if (tp + fn > 0) tp / (tp + fn) else 0
    f1 <- if (prec + rec > 0) 2 * prec * rec / (prec + rec) else 0
    
    tibble(
      Scheme = scheme_label,
      threshold = thresh,
      tp = tp, fp = fp, fn = fn,
      Precision = prec, Recall = rec, `F1-Score` = f1
    )
  })
}

analyze_review_band <- function(comparisons_df, lower_b, upper_b, upper_inclusive = TRUE, label = "") {
  band_df <- if (upper_inclusive) {
    comparisons_df %>% filter(match_score >= lower_b & match_score <= upper_b)
  } else {
    comparisons_df %>% filter(match_score >= lower_b & match_score < upper_b)
  }
  
  band_df <- band_df %>%
    left_join(validation_set, by = c("record_id_a" = "trs_record_id", "record_id_b" = "tea_record_id")) %>%
    mutate(is_match = replace_na(is_match, FALSE))
  
  n_total <- nrow(band_df)
  n_true <- sum(band_df$is_match)
  share_correct <- if (n_total > 0) n_true / n_total else NA_real_
  
  tibble(
    Review_Band_Config = label,
    Band_Range = paste0("[", lower_b, ", ", upper_b, ifelse(upper_inclusive, "]", ")")),
    Pairs_In_Band = n_total,
    True_Matches = n_true,
    False_Positives = n_total - n_true,
    Share_Correct = share_correct
  )
}

v. Limits of the Linked File and Sensitivity to the Answer Key

The linked file is not reality, it is guess that two records contain the same person.

  1. Whose records are more likely to be missed or linked wrongly: Under string-similarity scoring, single-character typos penalize short surnames much more heavily than 9-letter surnames, causing short-surname educators to be disproportionately dropped at high thresholds. Conversely, educators from demographic groups with high surname concentration face a much higher baseline collision rate, increasing false-positive risks when birthdates are missing or noisy. Furthermore, in real TEA/TRS files, women and younger educators who change surnames upon marriage or divorce will fail both surname blocking and surname similarity rules if historical name tables are unavailable.
  2. What the linked file cannot tell you: The linked file cannot distinguish whether an unlinked TRS member is genuinely absent from TEA or is a certified teacher whose record failed linkage due to a marital name change or multi-field corruption.
  3. What would change if the answer key were wrong or unavailable: In our simulation, we know with \(100\%\) confidence that TRS_1..TRS_100 correspond to TEA_1..TEA_100. In real administrative merges, no ground-truth answer key exists, or a “gold standard” training sample is constructed from clerical review of easy cases. If our answer key omitted hard-to-link cases (such as marital surname changes or multi-field errors), our evaluation would severely overstate recall and understate optimal review-band width. For our conclusions to hold without an answer key, the conditional independence assumption across fields given match status and the missing-at-random / homogeneous corruption assumption across school districts and demographic groups must hold (Enamorado, Fifield, and Imai 2019, 356–359)—assumptions that should be audited via clerical sampling and post-linkage uncertainty weighting.

6. Data Note & Academic Integrity Statement

Data Note (Section 6b)

  • Source: Simulated administrative datasets representing the Teacher Retirement System of Texas (TRS, \(N = 500\)) and the Texas Education Agency (TEA, \(N = 500\)), generated in R using set.seed(42) across three corruption rates (0.10, 0.20 baseline, and 0.40) applied to num_matches = 100 true overlapping records (TRS_1..TRS_100 paired with TEA_1..TEA_100).
  • Method:
    • Preprocessing: Standardized first and last names to lowercase ASCII letters (standardize_names()), extracted first initial (substr(first_name_std, 1, 1)), and standardized dates of birth to Date format.
    • Deterministic Rules: Sequential exact matching on (Rule 1) last_name_std + first_name_std + dob_std, (Rule 2) last_name_std + first_initial + dob_std, and (Rule 3) last_name_std + first_name_std with |dob_a - dob_b| <= 1 day, removing matched records after each rule via anti_join().
    • Blocking: Blocked residual unmatched records on the first letter of standardized surname (substr(last_name_std_a, 1, 1) == substr(last_name_std_b, 1, 1)), reducing baseline candidate pairs from \(407 \times 407 = 165,649\) to \(19,030\) (\(-88.51\%\)).
    • Weights & Scoring: Compared baseline intuition weights (0.40 / 0.30 / 0.30) against empirical Fellegi-Sunter \(\log(m/u)\) weights (0.1754 / 0.1803 / 0.6443) and unsupervised EM weights (0.1761 / 0.1818 / 0.6421) using Jaro string similarity (stringdist(..., method = "jw")) and step-function DOB similarity (\(1.0\) exact, \(0.95\) for \(\pm 1\) day, \(0.0\) otherwise).
    • Threshold: Calibrated automatic link threshold at \(\tau = 0.90\) under default weights and \(\tau = 0.95\) under \(\log(m/u)\) weights, with a manual review band of \([0.85, 0.95]\).
  • Selection (What the Simulation and Answer Key Leave Out): The simulation draws uniformly from 12 first names and 12 surnames (18 first names and 16 surnames in TEA) with uniform birthdates between 1950-01-01 and 1985-12-31, and corrupts at most one field per true match (either a 3-letter first-name abbreviation, a 1-character surname typo, or a \(\pm 1\) day DOB shift) while leaving all 400 true non-matches uncorrupted. It leaves out multi-field corruptions, marital or hyphenated surname changes, middle names/suffixes, nicknames that change the first initial (e.g., Robert -> Bob, William -> Bill), twin birthdates, missing/NULL fields, and longitudinal duplicate records within the same agency.
  • Known Gaps & Privacy Practice: Single-pass blocking on the first letter of last_name_std drops true matches whenever a typo occurs in the first character of a surname (3 unreachable matches at corruption_rate = 0.40). In accordance with Section 6c privacy standards, raw generated names and birthdates are suppressed from exported tables, which display only de-identified record IDs (TRS_*, TEA_*), similarity scores, and institutional attributes.
